                    WHAT HAPPENED IN CODE
================================================================================

Before (causing error):

  const { error: insertError } = await supabase
    .from('lesson_notes')
    .insert([
      {
        school_id: user?.school_id,
        teacher_id: user?.id,
        ...
        term_id: null,  // ❌ ERROR: NOT NULL constraint violation!
        ...
      },
    ])

After (fixed):

  const insertData: any = {
    school_id: user?.school_id,
    teacher_id: user?.id,
    ...
    // Don't include term_id - let it be NULL naturally
  }
  
  const { error: insertError } = await supabase
    .from('lesson_notes')
    .insert([insertData])  // ✅ term_id will be NULL, no error
