┌─────────────────────────────────────────────────────────────────────────────┐
│ LAYER 2: SAFE ASYNC OPERATION HOOKS                                         │
│ File: src/hooks/useAsyncEffect.ts                                           │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                              │
│   A. useAsyncEffect Hook                                                    │
│   ─────────────────────                                                     │
│                                                                              │
│   Before (UNSAFE):                                                          │
│   ┌──────────────────────────────────────────────┐                          │
│   │ useEffect(() => {                           │                          │
│   │   const load = async () => {                │                          │
│   │     const data = await fetchData()          │                          │
│   │     setData(data)  // ← Can crash!          │                          │
│   │   }                                          │                          │
│   │   load()                                     │                          │
│   │ }, [])                                       │                          │
│   │                                              │                          │
│   │ ❌ If fetchData() throws → CRASH            │                          │
│   └──────────────────────────────────────────────┘                          │
│                                                                              │
│   After (SAFE):                                                             │
│   ┌──────────────────────────────────────────────┐                          │
│   │ useAsyncEffect(async () => {                │                          │
│   │   const data = await fetchData()            │                          │
│   │   setData(data)  // ← Protected!            │                          │
│   │ }, [])                                       │                          │
│   │                                              │                          │
│   │ ✅ If fetchData() throws → Caught & logged  │                          │
│   └──────────────────────────────────────────────┘                          │
│                                                                              │
│                                                                              │
│   B. safePromiseAll Function                                                │
│   ──────────────────────────                                                │
│                                                                              │
│   Before (UNSAFE):                                                          │
│   ┌──────────────────────────────────────────────┐                          │
│   │ const [a, b, c] = await Promise.all([       │                          │
│   │   fetch1(),  ← If fails                     │                          │
│   │   fetch2(),                                 │                          │
│   │   fetch3(),                                 │                          │
│   │ ])                                           │                          │
│   │ // Entire operation CRASHES                 │                          │
│   │ ❌ One failure = complete failure           │                          │
│   └──────────────────────────────────────────────┘                          │
│                                                                              │
│   After (SAFE):                                                             │
│   ┌──────────────────────────────────────────────┐                          │
│   │ const [a, b, c] = await safePromiseAll([    │                          │
│   │   fetch1(),  ← If fails                     │                          │
│   │   fetch2(),                                 │                          │
│   │   fetch3(),                                 │                          │
│   │ ])                                           │                          │
│   │ // fetch1 returns null, others continue     │                          │
│   │ ✅ Partial success = still works            │                          │
│   └──────────────────────────────────────────────┘                          │
│                                                                              │
│                                                                              │
│   C. safeFetch Function (Auto-Retry)                                        │
│   ────────────────────────────────                                          │
│                                                                              │
│   Network error flow:                                                       │
│                                                                              │
│   User performs action                                                      │
│        ↓                                                                     │
│   First fetch attempt fails (network error)                                 │
│        ↓ (wait 500ms)                                                       │
│   Retry attempt 1 - fails                                                   │
│        ↓ (wait 1000ms)                                                      │
│   Retry attempt 2 - succeeds!                                               │
│        ↓                                                                     │
│   Data loaded, user happy                                                   │
│        ↓                                                                     │
│   ✅ Auto-recovery from network issues                                      │
│                                                                              │
│                                                                              │
│   D. withTimeout Function                                                   │
│   ───────────────────────                                                   │
│                                                                              │
│   Prevents hanging operations:                                              │
│                                                                              │
│   await withTimeout(fetchData(), 5000)                                      │
│        │                                                                     │
│        ├─ fetchData() completes in 2 sec → Returns data ✅                  │
│        │                                                                     │
│        └─ fetchData() hangs > 5 sec → Throws timeout error ✅               │
│           (won't hang forever)                                              │
│                                                                              │
└─────────────────────────────────────────────────────────────────────────────┘
